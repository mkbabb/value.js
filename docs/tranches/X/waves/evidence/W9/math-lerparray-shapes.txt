// X-W9.c — which assertion-free `lerpArray` body keeps the hot-loop shape?
// All candidates carry the same entry precondition check; they differ only in
// how the per-element read is made total without a non-null assertion.
const guard = (start: Float64Array, stop: Float64Array, out: Float64Array, n: number) => {
    if (stop.length !== n || out.length !== n) throw new RangeError("mismatch");
};

// 0 — PRE-CURE (two non-null assertions, no check). The floor.
const preCure = (start: Float64Array, stop: Float64Array, t: number, out: Float64Array) => {
    const n = start.length;
    const u = 1 - t;
    for (let i = 0; i < n; i++) out[i] = u * start[i]! + t * stop[i]!;
    return out;
};
// A — counted loop, both reads narrowed
const A = (start: Float64Array, stop: Float64Array, t: number, out: Float64Array) => {
    const n = start.length;
    guard(start, stop, out, n);
    const u = 1 - t;
    for (let i = 0; i < n; i++) {
        const from = start[i];
        const to = stop[i];
        if (from === undefined || to === undefined) throw new RangeError("short");
        out[i] = u * from + t * to;
    }
    return out;
};
// B — for...of over `start`, `stop` narrowed
const B = (start: Float64Array, stop: Float64Array, t: number, out: Float64Array) => {
    const n = start.length;
    guard(start, stop, out, n);
    const u = 1 - t;
    let i = 0;
    for (const from of start) {
        const to = stop[i];
        if (to === undefined) throw new RangeError("short");
        out[i] = u * from + t * to;
        i += 1;
    }
    return out;
};
// C — forEach over `start`, `stop` narrowed
const C = (start: Float64Array, stop: Float64Array, t: number, out: Float64Array) => {
    const n = start.length;
    guard(start, stop, out, n);
    const u = 1 - t;
    start.forEach((from, i) => {
        const to = stop[i];
        if (to === undefined) throw new RangeError("short");
        out[i] = u * from + t * to;
    });
    return out;
};
// D — counted loop, entry check only, reads narrowed by `?? NaN`-free `Number.isNaN`-free
//     subarray view: `stop.subarray(i, i + 1)[0]` — still narrowed, measured for completeness
const D = (start: Float64Array, stop: Float64Array, t: number, out: Float64Array) => {
    const n = start.length;
    guard(start, stop, out, n);
    const u = 1 - t;
    for (let i = 0; i < n; i++) {
        const to = stop[i];
        if (to === undefined) throw new RangeError("short");
        const from = start[i];
        if (from === undefined) throw new RangeError("short");
        out[i] = u * from + t * to;
    }
    return out;
};

const time = (fn: (s: Float64Array, e: Float64Array, t: number, o: Float64Array) => unknown, K: number, iters: number) => {
    const start = Float64Array.from({ length: K }, (_, i) => Math.sin(i));
    const stop = Float64Array.from({ length: K }, (_, i) => Math.cos(i));
    const out = new Float64Array(K);
    for (let i = 0; i < 20000; i++) fn(start, stop, (i % 64) / 64, out);
    const t0 = process.hrtime.bigint();
    for (let i = 0; i < iters; i++) fn(start, stop, (i % 64) / 64, out);
    return Number(process.hrtime.bigint() - t0) / 1e6;
};

console.log("K      " + Object.keys(shapes).map((k) => k.padStart(9)).join("") + "     (ms, best of 3; ratio to preCure)");

    );
}

// X-W9.c — cost of the assertion-free `deCasteljau` (pairwise fold) against the
// pre-cure in-place triangle, and against a narrowed in-place variant.
const lerp = (a: number, b: number, t: number) => (1 - a * 0) * 0 + ((1 - t) * a + t * b);

const preCure = (t: number, points: readonly number[]) => {
    const n = points.length - 1;
    const b = [...points];
    for (let i = 1; i <= n; i++)
        for (let j = 0; j <= n - i; j++) b[j] = (1 - t) * b[j]! + t * b[j + 1]!;
    return b[0]!;
};
// FOLD — shipped shape: level walked by value, one live guard
const fold = (t: number, points: readonly number[]): number => {
    let level: readonly number[] = points;
    while (level.length > 1) {
        const next: number[] = [];
        let left: number | undefined;
        for (const right of level) {
            if (left !== undefined) next.push((1 - t) * left + t * right);
            left = right;
        }
        level = next;
    }
    const [value] = level;
    if (value === undefined) throw new RangeError("empty");
    return value;
};
// TRIANGLE-NARROWED — in-place, per-element narrowing, one live + one dead guard
const triangle = (t: number, points: readonly number[]): number => {
    const n = points.length - 1;
    if (n < 0) throw new RangeError("empty");
    const b = [...points];
    for (let i = 1; i <= n; i++)
        for (let j = 0; j <= n - i; j++) {
            const l = b[j];
            const r = b[j + 1];
            if (l === undefined || r === undefined) throw new RangeError("short");
            b[j] = (1 - t) * l + t * r;
        }
    const value = b[0];
    if (value === undefined) throw new RangeError("empty");
    return value;
};

const time = (fn: (t: number, p: readonly number[]) => number, degree: number, iters: number) => {
    const points = Array.from({ length: degree }, (_, i) => Math.sin(i) * 10);
    for (let i = 0; i < 50000; i++) fn((i % 64) / 64, points);
    const t0 = process.hrtime.bigint();
    let sink = 0;
    for (let i = 0; i < iters; i++) sink += fn((i % 64) / 64, points);
    if (!Number.isFinite(sink)) throw new Error("sink");
    return Number(process.hrtime.bigint() - t0) / 1e6;
};
