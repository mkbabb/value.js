const isPalettesApi = (url: URL) =>
    !/\/(@fs|@id|@vite|node_modules)\//.test(url.pathname) &&
    !/\.\w+$/.test(url.pathname) &&
    /(^|\/)palettes(\/|$)/.test(url.pathname);

function luminance([r, g, b]: number[]): number {
    const lin = (c: number) => {
        const s = c / 255;
        return s <= 0.04045 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
    };
    return 0.2126 * lin(r!) + 0.7152 * lin(g!) + 0.0722 * lin(b!);
}
const ratio = (a: number[], b: number[]) => {
    const [hi, lo] = [luminance(a), luminance(b)].sort((x, y) => y - x);
    return (hi! + 0.05) / (lo! + 0.05);
};

async function chipRatio(
    browser: Browser,
    colorScheme: "light" | "dark",
    contrast: "no-preference" | "more",
): Promise<Record<string, unknown>> {
    const context = await browser.newContext({
        viewport: { width: 1440, height: 900 },
        colorScheme,
        contrast,
    });
    try {
        const page = await context.newPage();
        await page.addInitScript((t) => {
            try {
                localStorage.setItem("vueuse-color-scheme", t);
            } catch {
                /* private window */
            }
        }, colorScheme);
        await page.route(isPalettesApi, (route) =>
            route.request().method() === "GET"
                ? route.fulfill({
                      status: 200,
                      contentType: "application/json",
                      body: JSON.stringify({ data: WALL, nextCursor: null, hasMore: false }),
                  })
                : route.fulfill({ status: 200, contentType: "application/json", body: "{}" }),
        );
        await page.goto("/#/browse", { waitUntil: "domcontentloaded", timeout: 90_000 });
        const chip = page
            .locator('[role="article"].palette-card .palette-meta__chip')
            .filter({ visible: true })
            .first();
        await expect(chip).toBeVisible({ timeout: 90_000 });
        await page.waitForTimeout(1200);
        return await chip.evaluate((el) => {
            const rd = (e: Element, n: string) => getComputedStyle(e).getPropertyValue(n).trim();
            const pane = el.closest(".pane-wrapper");
            const root = document.documentElement;
            const names = ["--ink-primary", "--ink-muted", "--muted-foreground", "--foreground", "--on-glass-muted", "--glass-tint-strength"];
            const o: Record<string, unknown> = { color: getComputedStyle(el).color, inPane: !!pane };
            for (const n of names) { o["root" + n] = rd(root, n); if (pane) o["pane" + n] = rd(pane, n); }
            return o;
        });
    } finally {
        await context.close();
    }
}
for (const scheme of ["light", "dark"] as const) {

        for (const contrast of ["no-preference", "more"] as const)
            console.log(`N2-TOKENS ${scheme} ${contrast}`, JSON.stringify(await chipRatio(browser, scheme, contrast)));
    });
}
