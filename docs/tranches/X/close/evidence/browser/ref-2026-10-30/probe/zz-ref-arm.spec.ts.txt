import { test, type Page } from "@playwright/test";
const lin = (c: number) => { const v = c / 255; return v <= 0.04045 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4; };
const Y = ([r, g, b]: number[]) => 0.2126 * lin(r!) + 0.7152 * lin(g!) + 0.0722 * lin(b!);
const ratio = (a: number[], b: number[]) => { const [p, q] = [Y(a), Y(b)]; return (Math.max(p, q) + .05) / (Math.min(p, q) + .05); };
async function grid(page: Page, png: Buffer) {
    return page.evaluate(async (b64) => {
        const img = new Image(); img.src = `data:image/png;base64,${b64}`; await img.decode();
        const c = document.createElement("canvas"); c.width = img.width; c.height = img.height;
        const g = c.getContext("2d")!; g.drawImage(img, 0, 0);
        const out: number[][] = []; const sx = Math.max(1, img.width >> 4), sy = Math.max(1, img.height >> 2);
        for (let y = 0; y < img.height; y += sy) for (let x = 0; x < img.width; x += sx) out.push(Array.from(g.getImageData(x, y, 1, 1).data.slice(0, 3)));
        return out;
    }, png.toString("base64"));
}
async function arm(page: Page, label: string, sel: string) {
    const loc = page.locator(sel).first();
    await loc.waitFor({ timeout: 60_000 });
    const color = await loc.evaluate((e) => getComputedStyle(e).color);
    const ink = await page.evaluate((c) => { const x = document.createElement("canvas").getContext("2d")!; x.fillStyle = c; x.fillRect(0, 0, 1, 1); return Array.from(x.getImageData(0, 0, 1, 1).data.slice(0, 3)); }, color);
    const style = await page.addStyleTag({ content: `${sel},${sel} *{color:transparent!important;-webkit-text-fill-color:transparent!important;text-shadow:none!important}` });
    await page.waitForTimeout(150);
    const ground = await grid(page, await loc.screenshot({ animations: "disabled" }));
    await style.evaluate((n) => n.parentNode?.removeChild(n));
    const rs = ground.map((g) => ratio(ink, g));
    const ambient = await page.evaluate(() => getComputedStyle(document.documentElement).getPropertyValue("--ink-ambient-l").trim());
    console.log(`[ref-arm] ${label} ${JSON.stringify({ ambient, color, ink, worst: +Math.min(...rs).toFixed(2), best: +Math.max(...rs).toFixed(2), n: rs.length })}`);
}
for (const [name, seed] of [["default", null], ["deadband", "oklch(0.66 0.16 28)"]] as const) {
    test(`arm ${name}`, async ({ page }) => {
        test.setTimeout(200_000);
        await page.setViewportSize({ width: 1440, height: 900 });
        await page.emulateMedia({ colorScheme: "light", reducedMotion: "reduce" });
        await page.addInitScript(([css]) => {
            localStorage.setItem("vueuse-color-scheme", "light");
            if (css) localStorage.setItem("color-picker", JSON.stringify({ inputColor: css, savedColors: [] }));
        }, [seed] as const);
        await page.goto("/#/", { waitUntil: "load", timeout: 120_000 });
        await page.waitForTimeout(2500);
        await arm(page, `${name} component-name`, '[data-o18="component-name"]').catch((e) => console.log(`[ref-arm] ${name} component-name ABSENT ${String(e).slice(0, 80)}`));
        await page.goto("/#/browse", { timeout: 60_000 });
        await page.waitForTimeout(2500);
        await arm(page, `${name} browse caption`, "main .pane-header .pane-header-desc");
    });
}
