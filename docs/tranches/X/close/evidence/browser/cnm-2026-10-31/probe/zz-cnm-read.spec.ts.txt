// SERVED MODEL: claude-opus-5-5 — X.W11.cnm (COHESION §0gy.2): the oracle-grade read of the
// default seed's component-name ink. Copied into a scratch tree as e2e/smoke/zz-cnm-read.spec.ts.
// Ink: the computed colour realized on the 8-bit grid by a 1x1 canvas (§0gw.3) and, as a
// cross-check, the painted glyph pixels. Ground: the real composite under the glyphs, read from a
// screenshot with every glyph masked (color transparent, asserted at the DOM), sampled at EVERY
// pixel inside the text's own line boxes (Range client rects, 1 px inset). Verdict = the WORST
// plate sample. Three frames 400 ms apart; the worst over all three is reported.
import { test, expect, type Page } from "@playwright/test";
const lin = (c: number) => { const v = c / 255; return v <= 0.04045 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4; };
const Y = ([r, g, b]: number[]) => 0.2126 * lin(r!) + 0.7152 * lin(g!) + 0.0722 * lin(b!);
const ratio = (a: number[], b: number[]) => { const [p, q] = [Y(a), Y(b)]; return (Math.max(p, q) + 0.05) / (Math.min(p, q) + 0.05); };
const SEL = '[data-o18="component-name"]';

async function rects(page: Page) {
    return page.locator(SEL).first().evaluate((el) => {
        const r = document.createRange(); r.selectNodeContents(el);
        return Array.from(r.getClientRects()).filter((b) => b.width > 2 && b.height > 2).map((b) => ({ x: b.x, y: b.y, w: b.width, h: b.height }));
    });
}
async function pixels(page: Page, png: Buffer, boxes: Array<{ x: number; y: number; w: number; h: number }>) {
    return page.evaluate(async ([b64, bs]) => {
        const img = new Image(); img.src = `data:image/png;base64,${b64}`; await img.decode();
        const c = document.createElement("canvas"); c.width = img.width; c.height = img.height;
        const g = c.getContext("2d", { willReadFrequently: true })!; g.drawImage(img, 0, 0);
        const out: number[][] = [];
        for (const b of bs) {
            const x0 = Math.ceil(b.x) + 1, y0 = Math.ceil(b.y) + 1, x1 = Math.floor(b.x + b.w) - 1, y1 = Math.floor(b.y + b.h) - 1;
            const d = g.getImageData(x0, y0, x1 - x0, y1 - y0).data;
            for (let i = 0; i < d.length; i += 4) out.push([d[i]!, d[i + 1]!, d[i + 2]!]);
        }
        return out;
    }, [png.toString("base64"), boxes] as const);
}
async function settledInk(page: Page) {
    let last = "";
    for (let i = 0; i < 40; i++) {
        const now = await page.locator(SEL).first().evaluate((e) => getComputedStyle(e).color);
        if (now === last && now !== "") return now;
        last = now; await page.waitForTimeout(500);
    }
    throw new Error(`component-name ink never settled (last ${last})`);
}

for (const [scheme, motion] of [["light", "no-preference"], ["dark", "no-preference"], ["light", "reduce"]] as const) {
    test(`cnm default-seed component-name, ${scheme} ${motion}`, async ({ page }) => {
        test.setTimeout(180_000);
        await page.setViewportSize({ width: 1440, height: 900 });
        await page.emulateMedia({ colorScheme: scheme, reducedMotion: motion });
        await page.goto("/", { waitUntil: "load", timeout: 120_000 });
        await page.locator(SEL).first().waitFor({ timeout: 60_000 });
        await page.waitForTimeout(2500);
        const color = await settledInk(page);
        const ink = await page.evaluate((c) => { const x = document.createElement("canvas").getContext("2d")!; x.fillStyle = c; x.fillRect(0, 0, 1, 1); return Array.from(x.getImageData(0, 0, 1, 1).data.slice(0, 3)); }, color);
        const boxes = await rects(page);
        expect(boxes.length, "the name has line boxes").toBeGreaterThan(0);
        const frames: Array<{ worst: number; best: number; painted: number; n: number }> = [];
        for (let f = 0; f < 3; f++) {
            const painted = await pixels(page, await page.screenshot(), boxes);
            const style = await page.addStyleTag({ content: `${SEL},${SEL} *{color:transparent!important;-webkit-text-fill-color:transparent!important;text-shadow:none!important;transition:none!important}` });
            expect(await page.locator(SEL).first().evaluate((e) => getComputedStyle(e).color), "the glyphs are masked").toBe("rgba(0, 0, 0, 0)");
            await page.evaluate(() => new Promise((r) => requestAnimationFrame(() => requestAnimationFrame(() => r(null)))));
            const plate = await pixels(page, await page.screenshot(), boxes);
            await style.evaluate((n) => n.parentNode?.removeChild(n));
            const rs = plate.map((g) => ratio(ink, g));
            const paintedWorst = Math.min(...plate.map((g) => Math.max(...painted.map((p) => ratio(p, g)))));
            frames.push({ worst: Math.min(...rs), best: Math.max(...rs), painted: paintedWorst, n: rs.length });
            await page.waitForTimeout(400);
        }
        const worst = Math.min(...frames.map((f) => f.worst));
        const ambient = await page.evaluate(() => getComputedStyle(document.documentElement).getPropertyValue("--ink-ambient-l").trim());
        const reading = { scheme, motion, color, ink, ambient, boxes, frames, worst };
        test.info().annotations.push({ type: "cnm-reading", description: JSON.stringify(reading) });
        console.log(`[cnm] ${JSON.stringify(reading)}`);
        expect(worst, `component-name ${color} → ${ink} worst plate sample`).toBeGreaterThanOrEqual(4.5);
    });
}
