// SERVED MODEL: claude-opus-5-5
//
// value.js's CSS grammar — the SOURCE OF TRUTH (X.P.W6 · the owner's ruling 2026-09-23,
// "No custom grammar, unless it's BBNF"). X.P.W7: `@mkbabb/bbnf-lang`'s `bbnf gen` compiles these
// modules AHEAD OF TIME (`scripts/gen-grammar.mjs`) into `src/css/bbnf/generated/`, the checked-in
// parser value.js runs; `src/css/bbnf/actions.ts` gives it its semantic actions (values, never grammar).
//
// Each module `@import`s the modules whose rules it reads; `css.bbnf` is the root.
//
// tokens.bbnf — css-syntax-3 §4 tokens, as the value and colour productions read them.
// Keywords and function names are ASCII case-insensitive (`/…/i`); every keyword carries an
// ident-boundary lookahead so `nonesuch` is never `none`.

// A comment is consumed as whitespace is, and stands for nothing (css-syntax-3 §4.3.2): every component-
// value separator below reads `/* … */` wherever it reads whitespace (`auto /**/`, `1px/**/2px`); X.P.W8 `.v`. A `/`
// that opens a comment is never the slash separator; an unclosed comment is refused.
ws       = /\s*/ ;
ws1      = /\s+/ ;
// The stylesheet layer's readers split TEXT on whitespace alone (`stylesheet.bbnf`); a component value
// reads a comment where it reads whitespace: `sp` / `sp1`.
sp       = /(?:\s|\/\*[\s\S]*?\*\/)*/ ;
sp1      = /(?:\s|\/\*[\s\S]*?\*\/)+/ ;
comma    = /(?:\s|\/\*[\s\S]*?\*\/)*,(?:\s|\/\*[\s\S]*?\*\/)*/ ;
slash    = /(?:\s|\/\*[\s\S]*?\*\/)*\/(?!\*)(?:\s|\/\*[\s\S]*?\*\/)*/ ;
close    = /(?:\s|\/\*[\s\S]*?\*\/)*\)/ ;

// <number> — a bare number is never the prefix of a dimension or a percentage, and never the
// integer part of a longer number (`0` of `0.15deg`): css-syntax-3 §4.3.12 consumes digits greedily.
number     = /[+-]?(?:\d*\.\d+|\d+(?!\.\d))(?:[eE][+-]?\d+)?(?![\w%\\])/ ;
percentage = /[+-]?(?:\d*\.\d+|\d+)(?:[eE][+-]?\d+)?%/ ;
// A dimension and an angle capture their number and their unit: the leaf hands the pair to its
// `groups` action (`src/css/bbnf/math.ts`), so the token is split once, here, never re-split.
dimension  = /([+-]?(?:\d*\.\d+|\d+)(?:[eE][+-]?\d+)?)([a-zA-Z_][\w-]*)/ ;
angle      = /([+-]?(?:\d*\.\d+|\d+)(?:e[+-]?\d+)?)(deg|grad|rad|turn)(?![\w-])/i ;

none       = /none(?![\w-])/i ;
ident      = /-?[a-zA-Z_][\w-]*|--[\w-]*/ ;
dashedIdent = /--[\w-]+/ ;
string     = /"(?:\\[\s\S]|[^"\\])*"|'(?:\\[\s\S]|[^'\\])*'/ ;

// An opaque balanced run — the body of `var()`/`env()` fallbacks, whose grammar is the
// property's own and is checked only at computed-value time (css-variables-1 §3).
balanced   = ( /[^()"']+/ | string | "(" , balanced , ")" ) * ;
