// SERVED MODEL: claude-opus-5-5
//
// value.bbnf — component values (css-values-4 §2, css-syntax-3 §5.4.8), keyframe selectors
// (css-animations-1 §3.2 + scroll-animations-1 §4.2 named ranges) and <easing-function>
// (css-easing-2 §2–§4), as `/css`'s `parseCssValue` · `parseCssValues` · `parseCssScalar` ·
// `parseKeyframeSelector` · `parseTimingFunction` read them.

@import "tokens.bbnf" ;
@import "math.bbnf" ;
@import "color.bbnf" ;

// ── component values ──────────────────────────────────────────────────────────────
// Precedence, loosest first: the comma list, the slash list, the space list.
valueTop   = sp >> commaList << sp ;
commaList  = slashList , ( comma >> slashList ) * ;
slashList  = spaceList , ( slash >> spaceList ) * ;
spaceList  = valueTerm , ( termSep >> valueTerm ) * ;
// Juxtaposed terms are separated by whitespace, or stand directly against a `:`/`;` token, or are
// adjacent functions: css-syntax-3 §4 ends a function at its `)` token and the next `<function-token>`
// starts the next component with no whitespace between them (`rotate(0)scale(1.3)` —
// css-transforms-1 `<transform-list> = <transform-function>+`; X.P.W7 `.gap`). A `[]` block's
// `[` and `]` are tokens of their own (css-syntax-3 §4.3.1), so a grid `<line-names>` block stands
// directly against the components beside it: the minified `[a]1fr[b]`,
// `[top]auto[stage]1fr[bottom]auto` (css-grid-2 §7.2 `<track-list>`; X.P.W8 `.g`).
termSep    = /(?:\s|\/\*[\s\S]*?\*\/)+|(?=[:;])|(?<=[:;])|(?<=\))(?=[a-zA-Z_-][\w-]*\()|(?<=\])(?=[^\s,\/()\]:;])|(?<=[^\s,\/(\[:;])(?=\[)/ ;

valueTerm  = colorCall | mixCall | varCall | urlCall | urlString | typeCall | mathCall | call | numeric | string | lineNames | operator | identTerm | badTerm ;
// A component no production above reads (`@` in `calc(1px @ 2px)`). It is matched only to be
// REFUSED — its action answers `css_syntax` expecting a scalar, with the diagnostic spanning exactly
// this run — so a malformed value names its offending component rather than the whole input.
badTerm    = /[^\s(),\/:;"']+/ ;

// var() / env(): the fallback is a `<declaration-value>?` (css-variables-1 §2), which may end in a
// comma or be empty — `var(--a, red,)` and `var(--a,)` are valid; the trailing comma carries no item.
varCall    = /(?:var|env)(?=\()/i , /\((?:\s|\/\*[\s\S]*?\*\/)*/ >> varBody << close ;
varBody    = commaList << ( /\s*,/ ? ) ;

// A colour function is a colour, never a generic call: `rgb(1)` is a malformed colour.
colorCall  = relativeColor | rgbFn | hslFn | hwbFn | labFn | lchFn | oklabFn | oklchFn | colorFn | hex ;
colorHead  = /(?:rgba?|hsla?|hwb|lab|lch|oklab|oklch|color)\(/i ;
call       = ( callName - ( colorHead | /(?:url|type)\(/i ) ) , /\((?:\s|\/\*[\s\S]*?\*\/)*/ >> ( commaList ? ) << close ;
callName   = /(?:--[\w-]*|[a-zA-Z_-][\w-]*)(?=\()/ ;

// `color-mix()` (css-color-5 §3) in a value list, kept as the call it is (its colours as authored, never
// computed): `color-mix( <color-interpolation-method>? , [ <color> && <percentage [0,100]>? ]# )`. The method
// is the colour reader's own `mixMethod` (`in` a rectangular space, or a polar space with an optional
// `<hue-interpolation-method>`) and a comma ends it; the items are a comma list whose action checks each
// is one colour with at most one percentage, a literal one in [0, 100] (`src/css/bbnf/value.ts` `mixValue`),
// unchecked when the call holds an arbitrary substitution function (css-variables-1 §3). X.P.W8 `.t`.
mixCall    = /color-mix(?=\()/i , /\((?:\s|\/\*[\s\S]*?\*\/)*/ >> ( mixMethod << comma ) ? , commaList << close ;

// An unquoted url is ONE token, css-syntax-3 §4.3.6 `<url-token>`: `url(` not followed (after
// whitespace) by a quote reads its body whole — never as a component value list — up to its `)`,
// whitespace allowed only at either end; a quote, `(`, whitespace inside or a non-printable makes it
// a `<bad-url-token>` (§4.3.14), refused. `url("…")` is a `<function-token>` and stays a `call`.
// The body may be empty: `url()` is a `<url-token>` with empty contents (§4.3.6), valid to parse
// (it computes to an invalid image at use time); X.P.W7 `.gap2`.
urlHead    = /url\((?!\s*["'])/i ;
urlCall    = /url(?=\()/i , /\(\s*/ >> ( urlBody ? ) << close ;
urlBody    = /(?:[^\s"'()\\\x00-\x08\x0b\x0e-\x1f\x7f]|\\[^\n\r\f])+/ ;

// A quoted url is `url( <string> <url-modifier>* )` (css-values-4 §4.5): each modifier an identifier or
// a function. Its action refuses a modifier named twice (ASCII case-insensitively) and a request URL
// modifier whose argument is not its own (css-values-5 §4.5.1: `cross-origin(anonymous | use-credentials)`,
// `integrity(<string>)`, `referrer-policy(<keyword>)`). A quoted url this does not read is refused, never
// read as a generic call. X.P.W8 `.v`.
urlString  = /url(?=\()/i , /\(\s*/ >> string , ( sp1 >> ( call | identTerm ) ) * << close ;

// `type( <syntax> )` — attr()'s `<attr-type>` (css-values-5 §7.7) and `@function`'s parameter type, and
// `image-set()`'s `type( <string> )` (css-images-4 §2.2): one leaf whose two groups are the name and the
// argument, kept as authored. `<syntax> = '*' | <syntax-component> [ '|' <syntax-component> ]*`, a
// component a `<type-name>` or an identifier with an optional `+`/`#` multiplier (css-values-5 §8.1).
// The same leaf reads `element( <id-selector> )` (css-images-4 §2.4): a `<hash-token>` whose value is an
// identifier as `identTerm` reads one (css-syntax-3 §4.3.4 type flag "id"), kept as authored (`#a`,
// `#\31 a`); its action pairs each name with its own argument (`src/css/bbnf/value.ts` `leafCallValue`),
// and a generic `element()` call is refused (`src/css/bbnf/image.ts`). One alternation, not a rule of its
// own: a separate `elementCall` rule measured +79–89 B gzip against the E-6 ceiling, this +35. X.P.W8 `.i`.
typeCall   = /(type|element)\(\s*(#(?:[-_a-zA-Z\u0080-￿]|\\(?:[\da-fA-F]{1,6}[ \t\n\r\f]?|[^\n\r\f\da-fA-F]))(?:[-_a-zA-Z\d\u0080-￿]|\\(?:[\da-fA-F]{1,6}[ \t\n\r\f]?|[^\n\r\f\da-fA-F]))*|"(?:\\[\s\S]|[^"\\])*"|'(?:\\[\s\S]|[^'\\])*'|\*|(?:<transform-list>|(?:<[a-zA-Z][\w-]*>|-?[a-zA-Z_][\w-]*)[#+]?)(?:\s*\|\s*(?:<transform-list>|(?:<[a-zA-Z][\w-]*>|-?[a-zA-Z_][\w-]*)[#+]?))*)\s*\)/i ;

// `element( <id-selector> )` (css-images-4 §2.4): one leaf whose two groups are the name and the id
// selector, a `<hash-token>` whose value is an identifier (css-syntax-3 §4.3.4 type flag "id";
// selectors-4 §6.7), kept as authored — `#a`, `#-a`, `#\31 a`. Any other `element()` is refused
// (`src/css/bbnf/image.ts`). X.P.W8 `.i`.

// A math function (css-values-4 §10) reads its arguments as a call does, and admits a `()` simple
// block as a term: §10.1 `<calc-value> = … | ( <calc-sum> )` (`calc(50% - (1em / 2))`). The group is
// a call with the EMPTY name, which the serializer spells `( … )`; its body is never empty. A group
// stands only inside a math function or another group, never as a bare component value.
// Its arguments are then checked as a calculation — structure, arity, type (§10.1–§10.8, css-values-5
// `progress()`) — whatever property it stands in (`src/css/bbnf/calc.ts`; X.P.W8 `.v`).
mathCall   = /(?:calc|min|max|clamp|round|mod|rem|sin|cos|tan|asin|acos|atan2|atan|pow|sqrt|hypot|log|exp|abs|sign|progress)(?=\()/i ,
             /\((?:\s|\/\*[\s\S]*?\*\/)*/ >> mathComma << close ;
mathComma  = mathSlash , ( comma >> mathSlash ) * ;
mathSlash  = mathSpace , ( slash >> mathSpace ) * ;
mathSpace  = mathTerm , ( mathSep >> mathTerm ) * ;
// css-values-4 §10.1: `+` and `-` must be surrounded by whitespace; `*` and `/` need none. A `/` is
// already the slash list's separator; a `*` may stand directly against the terms on either side
// (`calc(1px*2)` reads as `calc(1px * 2)` does). `calc(1px+2px)` stays refused (X.P.W7 `.gap2`).
mathSep    = termSep | /(?:\s|\/\*[\s\S]*?\*\/)*(?=\*)|(?<=\*)/ ;
mathTerm   = group | valueTerm ;
group      = /\((?:\s|\/\*[\s\S]*?\*\/)*/ >> mathComma << close ;

// `1.` is not a number (css-syntax-3 §4.3.12, PB-12): a numeric is never followed by a `.`.
// Its two groups, the number and the unit, are the action's arguments (a `groups` leaf: no re-split).
numeric    = /([+-]?(?:\d*\.\d+|\d+)(?:[eE][+-]?\d+)?)([%a-zA-Z-]*)(?![\w\\.])/ ;
// A `-` that starts an identifier (`-zπ`, `-\31`) is not the operator (css-syntax-3 §4.3.10).
operator   = /<=|>=|==|!=|[+*<>=:;]|-(?![\w\u0080-\uffff\\-])/ ;
// An identifier's code points (css-syntax-3 §4.2 ident-start / ident code point, §4.3.11): ASCII
// letters, `_`, digits after the start, `-`, any non-ASCII code point (`-zπ`), and an escape — a hex
// escape of one to six digits with one optional whitespace it consumes (`\31 st`), or any other
// escaped code point (§4.3.7). The escape is kept as authored. X.P.W8 `.g`.
// Its first alternative reads a css-syntax-3 §7.1 `<urange>` (css-fonts-4 §4.5 `unicode-range`):
// `U+` (either case), one to six hex digits or `?` wildcards, and an optional `-` and one to six hex
// digits (`U+0025-00FF`, `U+4??`), kept as authored; its action (`src/css/bbnf/value.ts` `urangeValue`)
// refuses a non-trailing or ranged `?`, a start above the end and an end above U+10FFFF. One
// alternation, not a rule of its own: a separate `urange` rule measured +74 to +93 B gzip against the
// E-6 ceiling, this +26. X.P.W8 `.f`.
identTerm  = /[uU]\+[\da-fA-F?]{1,6}(?:-[\da-fA-F]{1,6})?(?![\w?\\-])|(?:[-_a-zA-Z\u0080-\uffff]|\\(?:[\da-fA-F]{1,6}[ \t\n\r\f]?|[^\n\r\f\da-fA-F]))(?:[-_a-zA-Z\d\u0080-\uffff]|\\(?:[\da-fA-F]{1,6}[ \t\n\r\f]?|[^\n\r\f\da-fA-F]))*(?![\w(\\-])/ ;
// A css-values-4 §4.2 `<custom-ident>`'s token, alone (css-syntax-3 §4.3.9 "would start an
// identifier"): `--`, or an optional `-` before an ident-start code point or an escape, then ident
// code points — `identTerm`'s identifier without its `<urange>` and with its start exact (a lone `-`
// and `-1` are not identifiers). An entry only: `syntax.ts` reads a `<custom-ident>` through it, never
// a string, a delimiter or a `[]` block. X.P.W8 `.x`.
customIdent = /(?:--|-?(?:[_a-zA-Z\u0080-\uffff]|\\(?:[\da-fA-F]{1,6}[ \t\n\r\f]?|[^\n\r\f\da-fA-F])))(?:[-_a-zA-Z\d\u0080-\uffff]|\\(?:[\da-fA-F]{1,6}[ \t\n\r\f]?|[^\n\r\f\da-fA-F]))*/ ;
// css-grid-2 §7.2 `<line-names> = '[' <custom-ident>* ']'` (`grid-template-columns: [rail] 1fr [end]`):
// a `[]` simple block (css-syntax-3 §5.4.8) holding only whitespace-separated identifiers — never a
// general `[]` block (`[1px]`, `[a, b]`, `["a"]`, `[[a]]` stay refused). One term, whose action spells
// it canonically, `[a b]` (X.P.W7 `.gap2`).
// Its identifiers take any non-ASCII code point as `identTerm`'s do (`[π]`; X.P.W8 `.g`).
lineNames  = /\[\s*(?:(?:-?[a-zA-Z_\u0080-\uffff]|--)[\w\u0080-\uffff-]*(?:\s+(?:-?[a-zA-Z_\u0080-\uffff]|--)[\w\u0080-\uffff-]*)*)?\s*\]/ ;

// A single scalar — `parseCssScalar`: a colour, a number with its unit, a string, an
// operator token or a keyword.
scalarTop  = sp >> scalarTerm << sp ;
scalarTerm = colorMix | lightDark | colorCall | numeric | string | operator | identTerm ;

// A whole colour — `parseCssColor`.
colorTop   = sp >> color << sp ;

// ── keyframe selectors ────────────────────────────────────────────────────────────
keyframeSelector = sp >> ( selectorKeyword | selectorNamed | percentage ) << sp ;
selectorKeyword  = /(?:from|to)(?![\w-])/i ;
selectorNamed    = /(?:entry|exit|cover|contain)(?![\w-])/i , ( sp1 >> percentage ) ? ;

// ── <easing-function> ─────────────────────────────────────────────────────────────
timingFunction = sp >> ( cubicBezier | stepsFn | linearFn | timingKeyword ) << sp ;
timingKeyword  = /(?:linear|ease-in-out|ease-in|ease-out|ease|step-start|step-end)(?![\w(-])/i ;
cubicBezier    = /cubic-bezier\(\s*/i >> number , comma >> number , comma >> number , comma >> number << close ;
stepsFn        = /steps\(\s*/i >> number , ( comma >> stepPosition ) ? << close ;
stepPosition   = /(?:jump-start|jump-end|jump-none|jump-both|start|end)(?![\w-])/i ;
linearFn       = /linear\(\s*/i >> linearStop , ( comma >> linearStop ) * << close ;
linearStop     = number , ( sp1 >> percentage ) ? , ( sp1 >> percentage ) ? ;
